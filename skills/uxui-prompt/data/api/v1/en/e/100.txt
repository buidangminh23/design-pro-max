Element: Folding flip timer
ID: 100 · Category: Widgets & Charts · Tags: timer, flip-clock, countdown, pomodoro, 3d
What it does: The top half of each digit card folds forward from the center hinge and the bottom half follows, unfolding into the next number. Only the digits that change flip; in the last 10 seconds the digits turn orange, and when it ends the cards shake once and a restart button appears.
How to use it: Pick 15s and press Start to watch the last 10 seconds and the finish. “+1m” adds a minute even while it runs
Footprint: fixed size, square
Needs: tokens.css (the --ux-* theme tokens and the .ux base). Map the tokens to the project's theme variables.
Live demo: https://lucianlabs.dev/uxui-prompt/#100

<script>
(function () {
  var root = document.currentScript.previousElementSibling;
  var clk = root.querySelector('.u100-clk'), st = root.querySelector('.u100-st'), pr = root.querySelector('.u100-pr i'), ps = root.querySelector('.u100-ps'), go = root.querySelector('.u100-go'), rs = root.querySelector('.u100-rs');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var PLAY = '<svg viewBox="0 0 16 16" aria-hidden="true"><path d="M4 2.4v11.2L13.2 8z"/></svg>', PAUSE = '<svg viewBox="0 0 16 16" aria-hidden="true"><rect x="3.4" y="2.6" width="3.2" height="10.8" rx="1"/><rect x="9.4" y="2.6" width="3.2" height="10.8" rx="1"/></svg>';
  var cards = [];
  [0, 1, 'c', 2, 3].forEach(function (k) {
    if (k === 'c') { var c = document.createElement('span'); c.className = 'u100-co'; c.setAttribute('aria-hidden', 'true'); c.innerHTML = '<i></i><i></i>'; clk.appendChild(c); return; }
    var f = document.createElement('span');
    f.className = 'u100-fl'; f.setAttribute('aria-hidden', 'true');
    f.innerHTML = '<span class="u100-t"><b>0</b></span><span class="u100-b"><b>0</b></span><span class="u100-ft"><b>0</b></span><span class="u100-fb"><b>0</b></span>';
    clk.appendChild(f);
    cards.push({ el: f, v: '0', t: f.querySelector('.u100-t b'), b: f.querySelector('.u100-b b'), ft: f.querySelector('.u100-ft'), fb: f.querySelector('.u100-fb') });
  });
  function flip(c, v) {
    if (c.v === v) return;
    var old = c.v; c.v = v;
    c.t.textContent = v;
    c.ft.firstChild.textContent = old; c.fb.firstChild.textContent = v;
    if (rm) { c.b.textContent = v; return; }
    c.b.textContent = old;
    c.ft.getAnimations().forEach(function (a) { a.cancel(); }); c.fb.getAnimations().forEach(function (a) { a.cancel(); });
    c.ft.animate([{ transform: 'perspective(320px) rotateX(0deg)', filter: 'brightness(1)' }, { transform: 'perspective(320px) rotateX(-90deg)', filter: 'brightness(.6)' }], { duration: 240, easing: 'cubic-bezier(.55,0,.9,.6)' });
    var a = c.fb.animate([{ transform: 'perspective(320px) rotateX(90deg)', filter: 'brightness(1.5)' }, { transform: 'perspective(320px) rotateX(0deg)', filter: 'brightness(1)' }], { duration: 300, delay: 240, easing: 'cubic-bezier(.2,.9,.3,1.25)', fill: 'forwards' });
    a.onfinish = function () { c.b.textContent = c.v; a.cancel(); };
  }
  var PRE = [[15, '15s'], [60, '1m'], [300, '5m'], [1500, '25m']];
  var total = 60, left = 60000, end = 0, run = false, timer = 0, last = '', sessions = 2;
  var pb = PRE.map(function (p) {
    var b = document.createElement('button');
    b.type = 'button'; b.textContent = p[1]; b.setAttribute('aria-pressed', String(p[0] === 60));
    b.addEventListener('click', function () { setTotal(p[0]); pb.forEach(function (x) { x.setAttribute('aria-pressed', String(x === b)); }); });
    ps.appendChild(b);
    return b;
  });
  var more = document.createElement('button');
  more.type = 'button'; more.className = 'more'; more.textContent = '+1m';
  ps.appendChild(more);
  function show(ms, quiet) {
    var s = Math.ceil(ms / 1000), str = ('0' + Math.floor(s / 60)).slice(-2) + ('0' + s % 60).slice(-2);
    if (str.length > 4) str = '9959';
    str.split('').forEach(function (ch, i) { if (quiet) { cards[i].v = ch; cards[i].t.textContent = ch; cards[i].b.textContent = ch; } else flip(cards[i], ch); });
    pr.style.setProperty('--p', Math.max(0, Math.min(1, ms / (total * 1000))).toFixed(4));
    root.classList.toggle('warn', run && s <= 10 && s > 0);
    if (str !== last) { clk.setAttribute('aria-label', 'Time left ' + Math.floor(s / 60) + ' min ' + s % 60 + ' sec'); last = str; }
    if (run && s <= 10 && s > 0) st.textContent = s + ' sec left';
  }
  function label() { go.innerHTML = (run ? PAUSE + 'Pause' : PLAY + (root.classList.contains('done') ? 'Restart' : left < total * 1000 ? 'Resume' : 'Start')); }
  function loop() {
    timer = 0;
    if (!root.isConnected || !run) return;
    left = Math.max(0, end - performance.now());
    show(left);
    if (left <= 0) return finish();
    timer = setTimeout(loop, 200);
  }
  function finish() {
    run = false; root.classList.remove('run', 'warn'); root.classList.add('done');
    st.textContent = 'Done · take a 5-min break';
    var dots = root.querySelectorAll('.u100-ss i'), n = root.querySelectorAll('.u100-ss i.on').length;
    if (n >= dots.length) { dots.forEach(function (d) { d.classList.remove('on'); }); n = 0; }
    dots[n].classList.add('on', 'pop'); setTimeout(function () { dots[n].classList.remove('pop'); }, 700);
    sessions++; root.querySelector('.u100-sn').textContent = sessions + ' focus sessions today';
    if (!rm) { clk.classList.remove('shake'); void clk.offsetWidth; clk.classList.add('shake'); }
    label();
  }
  function setTotal(sec) {
    clearTimeout(timer); run = false; root.classList.remove('run', 'warn', 'done');
    total = sec; left = sec * 1000;
    st.textContent = 'Ready';
    show(left); label();
  }
  go.addEventListener('click', function () {
    if (run) {
      run = false; clearTimeout(timer); left = Math.max(0, end - performance.now());
      root.classList.remove('run'); st.textContent = 'Paused'; label();
      return;
    }
    if (root.classList.contains('done')) { root.classList.remove('done'); left = total * 1000; show(left); }
    run = true; end = performance.now() + left;
    root.classList.add('run'); st.textContent = 'Focusing';
    label(); loop();
  });
  more.addEventListener('click', function () {
    if (root.classList.contains('done')) root.classList.remove('done');
    total += 60;
    if (run) end += 60000; else left += 60000;
    if (!run) st.textContent = 'Added 1 min';
    show(run ? end - performance.now() : left); label();
    more.animate([{ transform: 'scale(.92)' }, { transform: 'none' }], { duration: 260, easing: 'cubic-bezier(.3,1.4,.5,1)' });
  });
  rs.addEventListener('click', function () {
    var i = pb.findIndex(function (b) { return b.getAttribute('aria-pressed') === 'true'; });
    setTotal(i >= 0 ? PRE[i][0] : 60);
  });
  show(left, true); label();
})();
</script>
